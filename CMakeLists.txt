cmake_minimum_required(VERSION 3.10)
project(GuruxCompressionCpp LANGUAGES CXX)
option(COMPRESSION_TRACE "Compile V.44 trace callbacks and diagnostic snapshots" OFF)
set(GX_COMPRESSION_WITH_TRACE ${COMPRESSION_TRACE})
configure_file(development/include/GXCompressionConfig.h.in generated/GXCompressionConfig.h)
set(GX_COMPRESSION_EXTERNAL_BUFFER_HEADER "" CACHE STRING "Absolute full DLMS GXByteBuffer header; application supplies its implementation")
set(GX_COMPRESSION_BUFFER_LIBRARIES "" CACHE STRING "Libraries providing the external CGXByteBuffer")
add_library(gurux_compression development/src/GXBitArray.cpp development/src/GXV44Options.cpp development/src/GXV44Encoder.cpp development/src/GXV44Decoder.cpp)
# Use the same fixed configuration for the build tree and installed consumers.
target_compile_definitions(gurux_compression PUBLIC GX_COMPRESSION_CONFIG_HEADER="${CMAKE_CURRENT_BINARY_DIR}/generated/GXCompressionConfig.h")
if(COMPRESSION_TRACE)
    target_compile_definitions(gurux_compression PUBLIC COMPRESSION_TRACE)
endif()
if(GX_COMPRESSION_EXTERNAL_BUFFER_HEADER)
    target_compile_definitions(gurux_compression PUBLIC GX_COMPRESSION_EXTERNAL_BUFFER_HEADER="${GX_COMPRESSION_EXTERNAL_BUFFER_HEADER}")
    target_link_libraries(gurux_compression PUBLIC ${GX_COMPRESSION_BUFFER_LIBRARIES})
else()
    target_sources(gurux_compression PRIVATE development/src/GXByteBuffer.cpp)
endif()
target_include_directories(gurux_compression PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/development/include> $<INSTALL_INTERFACE:include/gurux-compression>)
set(GX_COMPRESSION_CXX_STANDARD 98)
if(GX_COMPRESSION_EXTERNAL_BUFFER_HEADER)
    # The supplied full DLMS header includes <cstdint>; standalone remains C++98.
    set(GX_COMPRESSION_CXX_STANDARD 11)
    target_compile_features(gurux_compression PUBLIC cxx_std_11)
    message(STATUS "External full DLMS buffer selected: C++11 for its header; standalone is C++98")
endif()
set_target_properties(gurux_compression PROPERTIES CXX_STANDARD ${GX_COMPRESSION_CXX_STANDARD} CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
    target_compile_options(gurux_compression PRIVATE /W4)
else()
    target_compile_options(gurux_compression PRIVATE -Wall -Wextra -pedantic)
endif()
include(GNUInstallDirs)
if(NOT GX_COMPRESSION_EXTERNAL_BUFFER_HEADER)
    install(TARGETS gurux_compression ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(DIRECTORY development/include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/gurux-compression FILES_MATCHING PATTERN "*.h" PATTERN "GXCompressionConfig.h" EXCLUDE)
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/generated/GXCompressionConfig.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/gurux-compression)

else()
    message(STATUS "External buffer: standalone installation disabled; use configured CMake target")
endif()
